import json
import google.generativeai as genai


path = "dictionary.json"
with open(path, "r", encoding="utf-8") as file:
    dictionary = json.load(file)



genai.configure(api_key="ppp")

def generate_sentence(prompt):
    """Use Gemini API to generate a well-formed sentence."""
    try:
        model = genai.GenerativeModel("gemini-2.5-flash")
        response = model.generate_content(prompt)
        return response.text.strip() if response else "Sentence formation failed."
    except Exception as e:
        print(f"Error generating sentence: {e}")
        return "Sentence formation failed."


def translate_word_or_sentence(dictionary, halegannada_text):
    words = halegannada_text.split()
    kannada_translations = []

    
    for word in words:
        kannada_translation = dictionary.get(word, word) 
        if isinstance(kannada_translation, list):  
            kannada_translations.append(' or '.join(kannada_translation))
        else:
            kannada_translations.append(kannada_translation)

    
    kannada_sentence = " ".join(kannada_translations)
    
    print(f"JSON Mapping (Hosakannada): {kannada_sentence}")
    
    kannada_prompt = f"Refine this Kannada sentence to be more grammatically correct: {kannada_sentence}"
    refined_kannada_sentence = generate_sentence(kannada_prompt)

    
    english_prompt = f"Translate this Kannada sentence to English with proper grammar: {refined_kannada_sentence}"
    english_sentence = generate_sentence(english_prompt)

    return refined_kannada_sentence, english_sentence


def main():
    halegannada_input = input("Enter a Halegannada word/sentence: ").strip()

    
    kannada_translation, english_translation = translate_word_or_sentence(dictionary, halegannada_input)

    
    print(f"Kannada (Hosagannada) translation: {kannada_translation}")
    print(f"English translation: {english_translation}")


if __name__ == "__main__":
    main()
